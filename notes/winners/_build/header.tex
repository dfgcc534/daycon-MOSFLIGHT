% --- blockquote 를 콜아웃 박스로 (💡 블록 강조) ---
\usepackage[most]{tcolorbox}
\tcbuselibrary{breakable}
% 프린트 적합: 배경은 흰색, 좌측 굵은 막대 + 얇은 테두리로만 구분 (잉크 최소)
\renewenvironment{quote}{%
  \begin{tcolorbox}[breakable,colback=white,colframe=black!45,boxrule=0.4pt,arc=1pt,leftrule=2.2pt,left=6pt,right=5pt,top=3pt,bottom=3pt]%
}{\end{tcolorbox}}
% 표가 페이지를 넘어가도 깨지지 않게
\usepackage{longtable}
% 가독성: 양끝맞춤(justify) 대신 왼쪽맞춤 → 글자간 간격 들쭉날쭉 제거
\usepackage{ragged2e}
\AtBeginDocument{\RaggedRight}
% 표는 작은 글씨 + 좁은 칸 간격 + 셀 내 왼쪽맞춤 → 칸 넘침/겹침 방지
\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\footnotesize\setlength{\tabcolsep}{4pt}\RaggedRight}
\AtBeginEnvironment{tabular}{\footnotesize\setlength{\tabcolsep}{4pt}}
\setlength{\emergencystretch}{3em}
